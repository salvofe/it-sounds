# Plancia di gioco — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the panel-based game screen with a card-table board (opponent across, stage zones with real cards, fanned hand, centre band, info drawer) without changing any game rule.

**Architecture:** New presentational components in `components/board/` read the existing Zustand store (`store/gameStore.ts`); the engine (`lib/engine`) is untouched. One `Card` component renders every card in three sizes using the existing skins (`lib/skins.ts`). A headless-Chrome check script (`scripts/board-check.mjs`) runs one check per player action and per layout requirement; each task turns its own checks green.

**Tech Stack:** Next.js 15, React 19, TypeScript (strict), Zustand, plain CSS in `app/globals.css`, Node 24 + headless Chrome over the DevTools protocol for checks.

**Spec:** `docs/superpowers/specs/2026-10-29-plancia-design.md`

## Global Constraints

- Rules and engine do not change: nothing under `lib/engine/` is edited; `npm test` (54 tests) stays green.
- On desktop from 1280×800 up the whole match fits one screen without vertical scrolling; at 390px wide it is usable (stacked, may scroll vertically, no horizontal page scroll).
- Readability rules: 'Press Start 2P' only for short labels (1–3 words) and never below 9px; text meant to be read in 'VT323' at 20px or more.
- Card looks come only from `lib/skins.ts` (`skinFor`, `cardStyles`, `cardStats`, `triggersGenreBonus`, `TYPE_COLORS`); the owner's genre picks the skin (yours for your cards, the AI's for its cards).
- UI copy in English.
- Every player action available today stays available (spec "Parità delle azioni").
- The `data-testid` names listed in Task 1 are a contract between components and the check script; do not rename them.

'use client'

import Card from './Card'

import type { Card as CardT, StageKey, StageSlot, StageState } from '@/types/game'

const ORDER: StageSlot[] = ['stage', 'artist1', 'artist2', 'artist3', 'hardware', 'track', 'track2', 'track3', 'guest']

// One stage on the table: its cards stacked, score, defences; yours can be focused,
// the AI's become targets in attack mode.
export default function StageZone({ owner, sk }: { owner: 'me' | 'ai'; sk: StageKey }) {
  const st = useGameStore()
  const s: StageState = owner === 'me' ? st.playerStages[sk] : st.aiStages[sk]
  const genre = owner === 'me' ? st.playerGenre : st.aiGenre
  const locked = owner === 'me' && !st.unlockedStages[sk]
  const focused = owner === 'me' && st.focusedStage === sk
  const target = owner === 'ai' && st.targetMode
  const passes = s.threshold > 0 && s.current >= s.threshold
  const cards = ORDER.flatMap(sl => (s[sl] ? [{ sl, card: s[sl] as CardT }] : []))

  function activate() {
    if (owner === 'ai') { if (st.targetMode) st.aiClick(sk); return }
    if (locked) return
    if (st.moveMode) { st.dropOnStage(sk); return }
    st.setFocus(sk)
  }
  const name = owner === 'me' ? sk.toUpperCase() : `AI ${sk.toUpperCase()}`
  const score = s.threshold > 0 ? `${s.current.toFixed(1)}k / ${s.threshold.toFixed(1)}k${passes ? ' ✓' : ''}${s.boost > 0 ? ` (+${s.boost}k boost)` : ''}`
    : locked ? 'Opens in round 2' : owner === 'me' ? 'Drop a STAGE here' : 'Empty'
  const headLabel = owner === 'ai'
    ? (target ? `Attack ${name}` : `${name}, ${score}`)
    : (locked ? `${name} locked` : st.moveMode ? `Move card to ${name}` : `Focus ${name}, ${score}`)

  return (
    <div data-testid={`zone-${owner}-${sk}`}
      className={`zone zone-${owner}${focused ? ' focused' : ''}${target ? ' target' : ''}${passes ? ' passes' : ''}${locked ? ' locked' : ''}`}
      onClick={activate}>
      <button type="button" className="zone-head" aria-label={headLabel} aria-pressed={owner === 'me' ? focused : undefined}
        disabled={locked || (owner === 'ai' && !target)} onClick={e => { e.stopPropagation(); activate() }}>
        <span className="zone-label">{name}{focused ? ' · FOCUS' : ''}</span>
        {(s.protected || s.decoy) && <span className="zone-shield">{s.protected ? 'SHIELD' : 'DECOY'}</span>}
        <span className="zone-score num">{score}</span>
      </button>
      <div className="zone-cards">
        {cards.map(({ sl, card }) => (
          <FieldSlot key={sl} owner={owner} sk={sk} sl={sl} card={card} genre={genre}
            moving={!!st.moveMode && st.moveMode.sk === sk && st.moveMode.sl === sl} />
        ))}
      </div>
      {s.combos.length > 0 && <div className="zone-combos">⚡ {s.combos.map(c => c.name).join(' + ')}</div>}
      {owner === 'me' && st.shop && s.stage && <ZoneShop sk={sk} s={s} />}
    </div>
  )

function FieldSlot({ owner, sk, sl, card, genre, moving }: { owner: 'me' | 'ai'; sk: StageKey; sl: StageSlot; card: CardT; genre: string; moving: boolean }) {

import { PHASES } from '@/lib/data'

  ARRANGEMENT: 'SOUNDCHECK ▶', SOUNDCHECK: 'START SHOW ▶', 'LIVE SHOW': 'CALCULATE ▶', RESULTS: 'NEW ROUND ▶',

const PHASE_SHORT: Record<Phase, string> = {

export default function CenterBand() {

  const phase = PHASES[st.phaseIdx] as Phase

  const win = RULES.short.win

  const maxRounds = win.kind === 'bestOf' ? win.maxRounds : 5
  const pips = (n: number) => '●'.repeat(n) + '○'.repeat(Math.max(0, need - n))

          ))}
        </ol>

            </>
