# Base viva Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A continuous backing track in the player's genre that gains a layer for every card on their stages, follows the round phases, and stays in time.

**Architecture:** Three pure modules in `lib/music/` (recipes = data, `mixFor` = game state → what should sound, `notesAt` = what sounds on one sixteenth) are unit-tested with Vitest; one Tone.js module (`player.ts`) turns them into audio and applies changes on the downbeat. The Zustand store feeds `mixFor(game, ME)` to the player after every commit. The rules engine is not touched.

**Tech Stack:** Next.js 15, React 19, TypeScript, Zustand, Tone.js ^14.8.49 (already installed), Vitest 3.

**Spec:** `docs/superpowers/specs/2026-10-29-base-viva-design.md` · approved prototype: `docs/superpowers/specs/assets/base-viva.html` (object `G` = the recipes, function `tick` = the sequencer).

## Global Constraints

- `lib/engine/` and the game rules do not change.
- No audio files and no new dependencies: synthesis with Tone.js only.
- Only the player's own cards (player `ME = 0`) make music; the AI's cards add nothing.
- Layer, phase and genre changes become live on the next downbeat; an attack glitch lands on the next sixteenth.
- No crackle: no instantaneous gain jumps (pump and glitch use ramps of 4–10 ms); minimum envelope attack 5 ms; convolution reverb (`Tone.Reverb`), not Freeverb; audio context with `latencyHint: 'playback'`; about −8 dB headroom before the limiter.
- Music sits below the sound effects (roughly −10 dB relative to them).
- If Tone.js fails to load or the audio context does not start, music stays off silently and the game continues.
- Mute (existing 🔊 button) silences music and effects; the Transport keeps running under mute.
- Code comments in English, matching the existing style (`// ── Section ───` headers, short comments).
- Every commit message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Work on branch `musica`.

## Review Focus

1. Many `startGame` calls in a row (the board check deals up to 40 games) → exactly one Transport repeat and one set of voices, never stacked loops. Pinned by M1 (`repeats === 1`).
2. `restartGame` then `startGame` within the 2 s fade-out → music comes back instead of being torn down by the pending stop. Pinned by M3.
3. Muting before the music starts → it starts muted. Pinned by M2.
4. A card leaving the field by a real engine action (pickup) → its layer goes away. Pinned by the `mixFor` pickup test in Task 2.
5. A round that ends in a draw, and the Classic mode (which has no `lastRoundWinner`) → `result` is still `'lose'` or `'win'` from the round scores. Pinned by the `mixFor` results tests in Task 2.

export const DEFAULT_GENRE = 'Synthwave'

export function recipeFor(genre: string): Recipe {
  return RECIPES[genre] ?? RECIPES[DEFAULT_GENRE]
}

// step → [scale degree, length in sixteenths]
export type Motif = Record<number, [number, number]>
const motifs = new Map<string, Motif>()

export function parseMotif(src: string): Motif {
  const hit = motifs.get(src)
  if (hit) return hit
  const m: Motif = {}
  for (const t of src.split(' ')) {
    const [s, d, l] = t.split(':').map(Number)
    m[s] = [d, l]
  }
  motifs.set(src, m)
  return m
}

// ── Base viva: what should sound ─────────────────────────────────────────────
// Pure: game state in, layer levels out. Only `me`'s own unlocked stages count;
// the opponent's cards never add layers (their attacks remove ours by changing state).

import { PHASES } from '@/lib/data'
import type { GameState, PlayerId } from '@/lib/engine'
import type { Phase, StageKey } from '@/types/game'
import { DEFAULT_GENRE, RECIPES } from './recipes'

export interface MusicMix {
  genre: string
  phase: Phase
  drums: 0 | 1 | 2 | 3
  artists: 0 | 1 | 2 | 3 | 4
  tracks: 0 | 1 | 2 | 3
  energy: 0 | 1 | 2 | 3
  sparkle: boolean
  result: 'win' | 'lose' | null
}

const KEYS: StageKey[] = ['s1', 's2', 's3']

export function mixFor(g: GameState, me: PlayerId): MusicMix {
  const p = g.players[me], opp = g.players[me === 0 ? 1 : 0]
  let drums = 0, artists = 0, tracks = 0, energy = 0, sparkle = false
  for (const k of KEYS) {
    if (!p.unlocked[k]) continue
    const s = p.stages[k]
    if (s.stage) drums++
    artists += [s.artist1, s.artist2, s.artist3].filter(Boolean).length
    tracks += [s.track, s.track2, s.track3].filter(Boolean).length
    if (s.hardware) energy++
    if (s.guest) sparkle = true
  }
  const phase = PHASES[g.phaseIdx]
  return {
    genre: RECIPES[p.genre] ? p.genre : DEFAULT_GENRE,
    phase,
    drums: Math.min(drums, 3) as MusicMix['drums'],
    artists: Math.min(artists, 4) as MusicMix['artists'],
    tracks: Math.min(tracks, 3) as MusicMix['tracks'],
    energy: Math.min(energy, 3) as MusicMix['energy'],
    sparkle,
    result: phase === 'RESULTS' ? (p.lastRoundScore > opp.lastRoundScore ? 'win' : 'lose') : null,
  }
}

// ── Base viva: one sixteenth of music ────────────────────────────────────────
// Pure port of the prototype's tick(): given a recipe, the live mix and the global
// step counter, list the notes to play. Durations are in sixteenths; the player
// converts to seconds and MIDI to note names.

import { CHORD, SCALE, parseMotif, type DrumVoice, type Recipe } from './recipes'
import type { MusicMix } from './mix'

export type VoiceId = 'base' | 'bass' | 'chords' | 'lead' | 'counter' | 'mel' | 'harm' | 'spk' | DrumVoice
export interface NoteEvent { voice: VoiceId; midi: number[]; dur: number; vel: number }
export interface StepOut { notes: NoteEvent[]; swing: number; pump: number | null }

// delay of each sixteenth within a beat, as a fraction of a sixteenth at full swing
const SWING_SHAPE = { '8n': [0, 0.333, 0.667, 0.333], '16n': [0, 0.5, 0, 0.5] } as const
export const PUMP_DEPTH = [1, 0.72, 0.55, 0.4]

const SILENT = { drums: 0, artists: 0, tracks: 0, energy: 0, sparkle: false }

function durAt(p: string, s: number): number {
  let n = 1
  while (s + n < 16 && p[s + n] === '-') n++
  return n
}

export function scaleNote(r: Recipe, deg: number): number {
  const sc = SCALE[r.scale], len = sc.length
  const o = Math.floor(deg / len), i = ((deg % len) + len) % len
  return r.root + 24 + sc[i] + 12 * o
}

export function notesAt(r: Recipe, mix: MusicMix, step: number, phraseStart: number, outro: boolean): StepOut {
  const s = step % 16, bar = Math.floor(step / 16) % 4
  const L = outro ? SILENT : mix
  const ch = r.prog[bar], nx = r.prog[(bar + 1) % 4]
  const root = r.root + ch[0], tones = CHORD[ch[1]].map(i => root + i)
  const notes: NoteEvent[] = []
  const add = (voice: VoiceId, midi: number[], dur: number, vel: number) => notes.push({ voice, midi, dur, vel })

  if (s === 0) add('base', [root + 12, root + 19], 15, 0.5)
  const dl = L.drums
  if (dl < 3 && s % 4 === 0) add('hat', [], 1, dl === 0 ? 0.3 : 0.18)
  r.parts.forEach((pt, i) => {
    if (i >= dl) return
    if (pt.p[s] === 'x') add(pt.v, [], 1, 0.95)
    else if (pt.p[s] === 'o') add(pt.v, [], 1, 0.4)
  })

  if (L.artists >= 1) {
    const p = r.v.bass.p, c = p[s]
    if (c !== '.' && c !== '-') {
      let n = root
      if (c === 'o') n = root + 12
      else if (c === 'f') n = root + 7
      else if (c === 'w') {
        const q = Math.floor(s / 4)
        n = [root, tones[1], tones[2], r.root + nx[0] - 1][q]
        while (n > r.root + 14) n -= 12
      }
      add('bass', [n], durAt(p, s), 0.9)
    }
  }
  if (L.artists >= 2) {
    const p = r.v.chords.p
    if (p[s] === 'x') add('chords', tones.map(t => t + 12), durAt(p, s), 0.7)
  }
  if (L.artists >= 3) {
    const c = r.v.lead.p[s]
    if (c >= '0' && c <= '9') {
      const i = Number(c)
      add('lead', [tones[i % tones.length] + 12 * Math.floor(i / tones.length) + 24], 1, 0.7)
    }
  }
  if (L.artists >= 4) {
    const p = r.v.counter.p
    if (p[s] === 'x') add('counter', [tones[1] + 24, tones[tones.length - 1] + 24], durAt(p, s), 0.55)
  }

  if (L.tracks >= 1) {
    const pos = phraseStart >= 0 ? (((step - phraseStart) % 64) + 64) % 64 : step % 64
    const motif = parseMotif(L.tracks >= 2 && pos >= 32 ? r.B : r.A)
    const m = motif[pos % 32]
    if (m) {
      add('mel', [scaleNote(r, m[0])], m[1], 0.85)
      if (L.tracks >= 3) add('harm', [scaleNote(r, m[0] - 2)], m[1], 0.7)
    }
  }

  if (L.sparkle) add('spk', [tones[s % tones.length] + 36], 0.5, 0.35)

  const swing = r.swing ? SWING_SHAPE[r.sub ?? '8n'][s % 4] * r.swing : 0
  const pump = L.energy >= 1 && s % 4 === 0 ? PUMP_DEPTH[L.energy] : null
  return { notes, swing, pump }
}

// ── Base viva: Tone.js player ────────────────────────────────────────────────
// The game says what it wants (apply); the player makes it live on the next
// downbeat so every layer enters in time. Safe to call before start: apply()
// queues, the rest no-op. Any failure leaves the game silent, never broken.

import { Sound } from '@/lib/sound'
import type { Phase } from '@/types/game'
import { CHORD, recipeFor, type Env, type Recipe, type Voice } from './recipes'
import { notesAt, type NoteEvent, type VoiceId } from './sequencer'
import type { MusicMix } from './mix'

type ToneModule = typeof import('tone')

const CUTOFF: Record<Phase, number> = { ARRANGEMENT: 850, SOUNDCHECK: 1700, 'LIVE SHOW': 16000, RESULTS: 3500 }
const LIVE_DB = -10   // music master in the Live Show; effects play on top
const PREP_DB = -13   // Arrangement, Soundcheck, Results
const DRIVE = [0, 0.12, 0.25, 0.4]
const DEFAULT_ENV: Env = { attack: 0.005, decay: 0.1, sustain: 0.2, release: 0.1 }

let T: ToneModule | null = null
let M: Record<string, any> | null = null
let V: Partial<Record<VoiceId, any>> = {}
let extras: any[] = []
let recipe: Recipe = recipeFor('Synthwave')
let built = ''
let want: MusicMix | null = null
let live: MusicMix | null = null
let step = 0, phraseStart = -1, barsInPhase = 0
let started = false, starting = false, muted = false
let repeatId: number | null = null
let stopTimer: ReturnType<typeof setTimeout> | null = null

// ── graph ───────────────────────────────────────────────────────────────────

function buildMaster(Tn: ToneModule) {
  const limiter = new Tn.Limiter(-2).toDestination()
  const mute = new Tn.Gain(muted ? 0 : 1).connect(limiter)
  const vol = new Tn.Volume(PREP_DB).connect(mute)
  const verb = new Tn.Reverb({ decay: 2.2, preDelay: 0.01 }).connect(vol)
  const filt = new Tn.Filter(CUTOFF.ARRANGEMENT, 'lowpass').connect(verb)
  const pump = new Tn.Gain(1).connect(filt)
  const drive = new Tn.Distortion(0.5).connect(pump)
  drive.wet.value = 0
  const bus = new Tn.Gain(1).connect(drive)
  const crashHp = new Tn.Filter(4500, 'highpass').connect(vol)
  const crash = new Tn.NoiseSynth({ noise: { type: 'white' }, envelope: { attack: 0.005, decay: 1.6, sustain: 0, release: 0.4 } }).connect(crashHp)
  crash.volume.value = -14
  const crusher = new Tn.BitCrusher(3).connect(vol)
  const glitch = new Tn.NoiseSynth({ noise: { type: 'pink' }, envelope: { attack: 0.005, decay: 0.04, sustain: 0, release: 0.01 } }).connect(crusher)
  glitch.volume.value = -8
  const sting = new Tn.PolySynth(Tn.Synth, { oscillator: { type: 'square' }, envelope: { attack: 0.005, decay: 0.2, sustain: 0.3, release: 0.6 } }).connect(verb)
  sting.volume.value = -14
  return { limiter, mute, vol, verb, filt, pump, drive, bus, crashHp, crash, crusher, glitch, sting }
}

function voice(Tn: ToneModule, spec: Voice, poly: boolean): any {
  const opts = { oscillator: { type: spec.osc }, envelope: spec.e ?? DEFAULT_ENV } as any
  const s: any = poly ? new Tn.PolySynth(Tn.Synth, opts) : new Tn.Synth(opts)
  if (spec.porta) s.portamento = spec.porta
  s.volume.value = spec.vol
  if (spec.dist) {
    const d = new Tn.Distortion(spec.dist).connect(M!.bus)
    s.connect(d)
    extras.push(d)
  } else s.connect(M!.bus)
  return s
}

function disposeVoices() {
  for (const v of Object.values(V)) { try { v.releaseAll?.(); v.dispose() } catch {} }
  for (const e of extras) { try { e.dispose() } catch {} }
  V = {}; extras = []
}

function buildVoices(genre: string) {
  const Tn = T!, r = recipeFor(genre)
  disposeVoices()
  recipe = r; built = genre
  Tn.Transport.bpm.value = r.bpm
  Tn.Transport.swing = 0          // swing is applied by the sequencer to the whole grid
  M!.verb.wet.value = r.verb
  V.base = voice(Tn, { ...r.v.base, e: { attack: 0.05, decay: 0.5, sustain: 0.8, release: 1.5 } }, true)
  const bass: any = new Tn.MonoSynth({
    oscillator: { type: r.v.bass.osc }, envelope: r.v.bass.e ?? DEFAULT_ENV,
    filter: { Q: 2, type: 'lowpass' },
    filterEnvelope: { attack: 0.005, decay: 0.25, sustain: 0.4, release: 0.2, baseFrequency: 140, octaves: 3.5 },
  } as any)
  bass.volume.value = r.v.bass.vol
  if (r.v.bass.porta) bass.portamento = r.v.bass.porta
  bass.connect(M!.bus)
  V.bass = bass
  V.chords = voice(Tn, r.v.chords, true)
  V.lead = voice(Tn, r.v.lead, false)
  V.counter = voice(Tn, r.v.counter, true)
  V.mel = voice(Tn, r.v.melody, false)
  V.harm = voice(Tn, { ...r.v.melody, vol: r.v.melody.vol - 5 }, false)
  V.spk = voice(Tn, { osc: r.v.sparkle.osc, e: { attack: 0.005, decay: 0.08, sustain: 0, release: 0.05 }, vol: r.v.sparkle.vol }, false)
  const kick: any = new Tn.MembraneSynth({ pitchDecay: 0.04, octaves: 6, envelope: { attack: 0.005, decay: r.kit.kick.decay, sustain: 0, release: 0.1 } } as any)
  kick.volume.value = r.kit.kick.vol
  kick.connect(M!.bus)
  V.kick = kick
  const snareHp = new Tn.Filter(r.kit.snare.hp, 'highpass').connect(M!.bus)
  const snare: any = new Tn.NoiseSynth({ noise: { type: 'white' }, envelope: { attack: 0.005, decay: r.kit.snare.decay, sustain: 0, release: 0.05 } } as any)
  snare.volume.value = r.kit.snare.vol
  snare.connect(snareHp)
  V.snare = snare
  const hatHp = new Tn.Filter(7000, 'highpass').connect(M!.bus)
  const hat: any = new Tn.NoiseSynth({ noise: { type: 'white' }, envelope: { attack: 0.005, decay: r.kit.hat.decay, sustain: 0, release: 0.02 } } as any)
  hat.volume.value = r.kit.hat.vol
  hat.connect(hatHp)
  V.hat = hat
  extras.push(snareHp, hatHp)
}

// ── timing ──────────────────────────────────────────────────────────────────

function stinger(time: number, win: boolean) {
  const Tn = T!, r = recipe.root + 24
  const run = win ? [0, 4, 7, 12, 16, 19] : [12, 8, 5, 3, 0]
  run.forEach((n, i) => M!.sting.triggerAttackRelease(Tn.Frequency(r + n, 'midi').toNote(), '8n', time + i * 0.11, 0.8))
  const chord = CHORD[win ? 'maj' : 'min'].map(x => Tn.Frequency(r + 12 + x, 'midi').toNote())
  M!.sting.triggerAttackRelease(chord, '2n', time + run.length * 0.11, 0.7)
}

function downbeat(time: number) {
  if (!want) return
  if (want.genre !== built) {
    buildVoices(want.genre)
    step = 0
    if (phraseStart >= 0) phraseStart = 0
  }
  const before = live
  live = { ...want }
  const phaseChanged = !before || before.phase !== live.phase
  barsInPhase = phaseChanged ? 0 : barsInPhase + 1
  if (live.tracks >= 1 && (!before || before.tracks === 0)) phraseStart = step
  if (live.tracks === 0) phraseStart = -1
  if (phaseChanged) {
    const cut = live.phase === 'RESULTS' ? (live.result === 'win' ? 5000 : 2200) : CUTOFF[live.phase]
    M!.filt.frequency.rampTo(cut, live.phase === 'LIVE SHOW' ? 0.05 : 1.2, time)
    M!.vol.volume.rampTo(live.phase === 'LIVE SHOW' ? LIVE_DB : PREP_DB, 0.5, time)
    if (before && live.phase === 'LIVE SHOW') M!.crash.triggerAttackRelease('1n', time, 0.9)
    if (live.phase === 'RESULTS' && live.result) stinger(time, live.result === 'win')
  }
  M!.drive.wet.rampTo(DRIVE[live.energy], 0.2, time)
  if (live.energy === 0) { M!.pump.gain.cancelScheduledValues(time); M!.pump.gain.setValueAtTime(1, time) }
}

function play(n: NoteEvent, time: number, d16: number) {
  const Tn = T!, v = V[n.voice]
  if (!v) return
  if (n.voice === 'kick') v.triggerAttackRelease(recipe.kit.kick.note, '8n', time, n.vel)
  else if (n.voice === 'snare' || n.voice === 'hat') v.triggerAttackRelease('32n', time, n.vel)
  else {
    const names = n.midi.map(m => Tn.Frequency(m, 'midi').toNote())
    v.triggerAttackRelease(names.length === 1 ? names[0] : names, d16 * n.dur * 0.95, time, n.vel)
  }
}

function tick(time0: number) {
  if (!T || !M) return
  if (step % 16 === 0) downbeat(time0)
  if (!live) { step++; return }
  const d16 = T.Time('16n').toSeconds()
  const outro = live.phase === 'RESULTS' && barsInPhase >= 1
  const out = notesAt(recipe, live, step, phraseStart, outro)
  const time = time0 + out.swing * d16
  if (out.pump !== null) {
    const g = M.pump.gain
    g.cancelScheduledValues(time)
    g.setValueAtTime(1, time)
    g.linearRampToValueAtTime(out.pump, time + 0.01)
    g.linearRampToValueAtTime(1, time + d16 * 3)
  }
  for (const n of out.notes) { try { play(n, time, d16) } catch { /* a dropped note never stops the show */ } }
  step++
}

function teardown() {
  if (stopTimer) { clearTimeout(stopTimer); stopTimer = null }
  const Tn = T
  if (Tn) {
    try { Tn.Transport.stop(); if (repeatId !== null) Tn.Transport.clear(repeatId) } catch {}
  }
  repeatId = null
  disposeVoices()
  if (M) for (const n of Object.values(M)) { try { n.dispose() } catch {} }
  M = null; built = ''; live = null; started = false
  step = 0; phraseStart = -1; barsInPhase = 0
}

// ── public API ──────────────────────────────────────────────────────────────

export const Music = {
  async start(): Promise<void> {
    if (typeof window === 'undefined') return
    if (stopTimer) teardown()          // a restart during the fade-out wins
    if (started || starting) return
    starting = true
    try {
      await Sound.unlock()
      const Tn = await import('tone')
      if (Tn.context.state !== 'running') await Tn.start()
      T = Tn
      M = buildMaster(Tn)
      step = 0; phraseStart = -1; barsInPhase = 0; live = null; built = ''
      repeatId = Tn.Transport.scheduleRepeat(tick, '16n')
      Tn.Transport.start('+0.1')
      started = true
    } catch (e) {
      console.warn('Music init failed', e)
      teardown()
    } finally {
      starting = false
    }
  },

  apply(mix: MusicMix): void {
    want = mix
  },

  glitch(): void {
    if (!started || !T || !M) return
    const now = T.Transport.nextSubdivision('16n'), d = T.Time('32n').toSeconds()
    for (let i = 0; i < 6; i++) {
      M.glitch.triggerAttackRelease('64n', now + i * d, 0.9)
      M.bus.gain.linearRampToValueAtTime(i % 2 ? 1 : 0.08, now + i * d + 0.004)
    }
    M.bus.gain.linearRampToValueAtTime(1, now + 6 * d + 0.004)
  },

  stop(afterMs = 0): void {
    if (!started || stopTimer) return
    stopTimer = setTimeout(() => {
      try { M?.vol.volume.rampTo(-60, 2) } catch {}
      stopTimer = setTimeout(() => { stopTimer = null; teardown(); want = null }, 2100)
    }, afterMs)
  },

  setMuted(m: boolean): void {
    muted = m
    try { M?.mute.gain.rampTo(m ? 0 : 1, 0.05) } catch {}
  },

  debug() {
    return { started, muted, step, repeats: repeatId === null ? 0 : 1, live, want }
  },
}
